// Month options for filter
const monthOptions = computed(() => {
    const months = []
    const currentDate = new Date()
    for (let i = 0; i < 12; i++) {
        const date = new Date(currentDate.getFullYear(), currentDate.getMonth() - i, 1)
        months.push({
            value: date.toISOString().slice(0, 7),
            label: date.toLocaleDateString('id-ID', { year: 'numeric', month: 'long' })
        })
    }
    return months
})

// Summary stats
const summaryStats = computed(() => {
    const total = integrationStats.value.reduce((acc, stat) => ({
        wellMed: acc.wellMed + stat.wellMedCount,
        satuSehat: acc.satuSehat + stat.satuSehatCount,
        unsigned: acc.unsigned + stat.unsignedCount,
        signed: acc.signed + stat.signedCount
    }), { wellMed: 0, satuSehat: 0, unsigned: 0, signed: 0 })

    const percentage = total.wellMed > 0 ? (total.satuSehat / total.wellMed * 100).toFixed(1) : 0

    return { ...total, percentage }
})
